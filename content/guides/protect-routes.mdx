# Protect Routes

## `requireAuth()` — session/user attach

```ts title="src/app/http/controllers/private.ts"
export const PrivateController = defineController({
  prefix: "/api/private",
  middleware: [requireAuth()],
  routes(api) {
    api.get("/", async (ctx) => {
      authorize(PrivatePolicy, ctx.user, "read");
      return ctx.store.private.all();
    });
  },
});
```

## `guestOnly({redirectTo})` — for public routes

```ts
export const AuthController = defineController({
  routes(api) {
    api.get("/signin", () => redirect("/"), {
      middleware: [guestOnly()],
    });
  },
});
```

## Page-level protection

```tsx
export default definePage({
  beforeLoad: async (ctx) => {
    const session = await getSession(ctx.request);
    if (!session) redirect("/signin");
  },
});
```

## Client usage

```tsx
const { session, user } = useSession();
```

> [!TIP]
> 401 = missing session, 403 = forbidden. Kwiva uses both.
