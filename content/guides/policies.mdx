# Add Policies

```ts title="src/app/policies/post.ts"
import { definePolicy, defineGate } from "@kwiva/auth";

export const PostPolicy = definePolicy({
  delete(user, post) { return user?.id === post.authorId; },
  edit(user, post) { return user?.id === post.authorId && !post.published; },
  read(user) { return user?.role === "admin" || true; },
});

export const PostGate = defineGate((user) => user?.role === "admin");
```

## Wire to routes

```ts
api.get("/:id", (ctx) => ctx.store.posts.find(ctx.params.id));
```

## Check via `ctx.can`

```ts
import { can } from "@kwiva/auth";
if (can(PostPolicy, ctx.user, "delete", post)) { /* allow */ }
```

## Gating on generated routes

`model.options.permission` — Kwiva checks `ctx.can` for each route.

> [!TIP]
> Add to `defineApp` via `policies: [PostPolicy]` — auto-discovery from `src/app/policies/`.
