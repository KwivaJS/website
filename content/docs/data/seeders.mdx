# Seeders

```ts title="src/database/seeders/posts-seed.ts"
import { defineSeeder, seed } from "@kwiva/data";
import { Post } from "@kwiva/data";

export const SeedPosts = defineSeeder(async ({ db }) => {
  await seed([{ key: "hello", value: { title: "Hello" } }], {
    // or upsert into Post
    Post: [{ id: 1, title: "Hello", status: "draft" }],
  });
});
```

`defineSeeder(handler: (ctx: SeederContext{db}) => void | Promise<void>)` — runs in order; `seed(entries, only?)` — idempotent.

CLI: `kwiva seed`, `kwiva seed <name>`.
